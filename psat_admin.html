📖 JSON 구조 가이드 — questions.json

앱은 번들에 포함된 questions.json을 읽어 문제를 표시합니다.
어드민에서 내보낸 파일을 Xcode에 교체하면 다음 배포 시 반영됩니다.

// questions.json 전체 구조 { "version": "1.0.0", // 스키마 버전 "exportedAt": "2025-09-07", // 내보낸 날짜 "questions": [ { "id": "deep_moi_2024_언어논리_01", // 고유 ID (자동생성) "examType": "deep_moi", // deep_moi | deep_na | basic_moi | basic_na "subject": "언어논리", // 헌법 | 언어논리 | 자료해석 | 상황판단 "year": 2024, // 출제연도 "number": 1, // 문번 "subtype": "강화·약화", // 세부 유형 (없으면 null) "difficulty": 3, // 1(하) ~ 5(상) "passage": "지문 내용...", // 지문 (없으면 null) "question": "문제 내용...", // 문제 본문 "choices": [ // 보기 배열 "① 첫 번째 보기", "② 두 번째 보기", "③ 세 번째 보기", "④ 네 번째 보기", "⑤ 다섯 번째 보기" ], "answer": 3, // 정답 번호 (1-based) "explanation": "해설 내용...", // 해설 (없으면 null) "passageImages": [ // 지문 이미지 파일명 배열 (순서 중요) "q_2024_자료해석_01_p1.png", "q_2024_자료해석_01_p2.png" ], "questionImages": [ // 문제 이미지 파일명 배열 (순서 중요) "q_2024_자료해석_01_q1.png" ], "choiceImages": { // 선지 이미지 — 키: 선지 번호(문자열), 값: 파일명 "2": "q_2024_자료해석_01_c2.png", // 2번 선지에 이미지 "4": "q_2024_자료해석_01_c4.png" // 4번 선지에 이미지 } } ] }
